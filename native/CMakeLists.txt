cmake_minimum_required(VERSION 3.20)
project(stsai_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED STS_ENGINE_DIR OR NOT EXISTS "${STS_ENGINE_DIR}/include/combat/BattleContext.h")
  message(FATAL_ERROR "Set STS_ENGINE_DIR to the locked sts_lightspeed checkout")
endif()
if(NOT DEFINED STSAI_ENGINE_REVISION)
  message(FATAL_ERROR "STSAI_ENGINE_REVISION is required; use scripts/build_native.py")
endif()
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${STS_ENGINE_DIR}/src/*.cpp")
pybind11_add_module(_lightspeed MODULE bridge.cpp ${ENGINE_SOURCES})
target_include_directories(_lightspeed PRIVATE "${STS_ENGINE_DIR}/include" "${STS_ENGINE_DIR}/json/single_include" "${STS_ENGINE_DIR}/json/include")
target_compile_definitions(_lightspeed PRIVATE STSAI_ENGINE_REVISION="${STSAI_ENGINE_REVISION}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(_lightspeed PRIVATE -O3 -Wno-shift-count-overflow)
endif()
option(STSAI_SANITIZE "Enable undefined-behavior checks (debug validation only)" OFF)
if(STSAI_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(_lightspeed PRIVATE -O1 -g -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(_lightspeed PRIVATE -fsanitize=undefined)
endif()
